function checkAnswers(){ const answers = { q1:"b", q2:"c", q3:"d", q4:"c", q5:"c", q6:"b", q7:"c", q8:"b", q9:"b", q10:"c", q11:"b", q12:"b", q13:"b", q14:"a", q15:"b", q16:"b", q17:"c", q18:"c", q19:"b", q20:"a", q21:"a", q22:"c", q23:"b", q24:"a", q25:"c", q26:"a", q27:"b", q28:"b", q29:"c", q30:"c" }; let score = 0; // Check answers for(let key in answers){ let selected = document.querySelector(`input[name="${key}"]:checked`); if(selected && selected.value === answers[key]){ score++; } } // Percentage let percentage = (score / 30) * 100; // Grade let grade = ""; if(percentage >= 90){ grade = "A+"; } else if(percentage >= 80){ grade = "A"; } else if(percentage >= 70){ grade = "B"; } else if(percentage >= 60){ grade = "C"; } else if(percentage >= 50){ grade = "D"; } else{ grade = "Fail"; } // Performance Message let message = ""; if(percentage >= 80){ message = "Excellent Work 🎉"; } else if(percentage >= 60){ message = "Good Job 👍"; } else{ message = "Need More Practice 📚"; } // Display Result document.getElementById("result").innerHTML = `

${message}

`; // Scroll to result document.getElementById("result").scrollIntoView({ behavior:"smooth" }); }
